用 `fetch` 追踪下载进度是完全可行的，但它比老式的 `XMLHttpRequest` 稍微复杂一些，因为它依赖于现代的 **Streams API**。

`fetch` 接收到响应后，其 **`response.body`** 属性是一个 **`ReadableStream`**（可读流），我们可以逐块读取这个流，并在读取过程中计算进度。

## 追踪下载进度的实现步骤 (使用 `async/await`)

这个方法要求浏览器支持 Streams API（现代浏览器基本都支持）。

### 步骤 1: 发起请求并获取总大小

你需要尝试从响应头部获取文件的总大小，这通常是通过 **`Content-Length`** 头部实现的。

```js
async function downloadWithProgress(url, onProgress) {
    const response = await fetch(url);
    
    if (!response.ok) {
        throw new Error(`HTTP 错误: ${response.status}`);
    }

    // 尝试获取总大小。注意：跨域请求可能没有这个头部。
    const contentLength = response.headers.get('Content-Length');
    const total = contentLength ? parseInt(contentLength, 10) : null;
    let loaded = 0;
    
    // ... 剩下的步骤 ...
}
```

### 步骤 2: 获取流读取器并循环读取

使用 `response.body.getReader()` 获取一个 **读取器（Reader）**，然后进入一个循环，每次调用 `reader.read()` 来获取下一块数据（chunk）

```js
// 1. 获取流读取器
    const reader = response.body.getReader();
    const chunks = []; // 用于存储所有读取到的数据块

    // 2. 循环读取数据
    while (true) {
        // reader.read() 返回一个 Promise，解析为 { done: boolean, value: Uint8Array }
        const { done, value } = await reader.read();

        if (done) {
            break; // 读取完成
        }

        // 3. 记录进度
        chunks.push(value);
        loaded += value.length;

        // 4. 调用回调函数，更新进度条
        if (total) {
            // onProgress 是一个回调函数，用于更新 UI
            onProgress(loaded, total); 
        } else {
            // 如果不知道总大小，只能显示已下载量
            onProgress(loaded, null); 
        }
    }
    
    // ... 最后的处理步骤 ...
```

### 步骤 3: 组合数据并返回结果

循环完成后，你需要将所有收集到的 **`Uint8Array`** 数据块合并成一个完整的响应体（例如 `Blob` 或 `ArrayBuffer`）。

```js
// 5. 将所有数据块合并成一个完整的 Uint8Array
    const allChunks = new Uint8Array(loaded);
    let offset = 0;
    for (const chunk of chunks) {
        allChunks.set(chunk, offset);
        offset += chunk.length;
    }

    // 6. 根据需要，将 Uint8Array 转换为 Blob (常用于文件) 或 Text (常用于 JSON/文本)
    const blob = new Blob([allChunks]);
    
    // 如果是 JSON 或文本：
    // const text = new TextDecoder('utf-8').decode(allChunks);
    // return JSON.parse(text); 
    
    return blob;
}
```

## 完整的可运行示例

这是一个集成了进度追踪和 UI 更新的完整函数。

```js
function updateProgress(loaded, total) {
    const percent = total ? Math.round((loaded / total) * 100) : '未知';
    console.log(`已下载: ${loaded} 字节 / 总共: ${total || '未知'} 字节 (${percent}%)`);
    
    // 实际项目中可以更新 HTML 元素，例如：
    // document.getElementById('progress-bar').style.width = percent + '%';
    // document.getElementById('progress-text').textContent = `${percent}%`;
}

async function downloadFileWithProgress(url) {
    console.log(`开始下载: ${url}`);
    
    const response = await fetch(url);
    if (!response.ok) {
        throw new Error(`下载失败，状态码: ${response.status}`);
    }

    // 1. 获取总文件大小 (Content-Length)
    const contentLength = response.headers.get('Content-Length');
    const total = contentLength ? parseInt(contentLength, 10) : null;
    let loaded = 0;

    // 2. 获取流读取器
    const reader = response.body.getReader();
    const chunks = [];

    // 3. 循环读取
    while (true) {
        const { done, value } = await reader.read();

        if (done) {
            break;
        }

        // 记录进度
        chunks.push(value);
        loaded += value.length;
        
        // 更新 UI
        updateProgress(loaded, total);
    }
    
    // 4. 合并数据块
    const allChunks = new Uint8Array(loaded);
    let offset = 0;
    for (const chunk of chunks) {
        allChunks.set(chunk, offset);
        offset += chunk.length;
    }

    console.log('下载完成！');
    // 返回最终结果 (例如，Blob 对象)
    return new Blob([allChunks]);
}
```
